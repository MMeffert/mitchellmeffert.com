```javascript
//Portfolio Filter
ElvishApp.prototype.initPortfolioFilter = function() {
    var grid = document.querySelector('.work-filter');
    if (!grid) return;

    // Initialize Isotope with vanilla JS
    var iso = new Isotope(grid, {
        itemSelector: '.col-lg-4',
        layoutMode: 'masonry',
        filter: '*',
        percentPosition: true,
        transitionDuration: '0.75s'
    });

    // Filter button handling with event delegation
    var filterButtons = document.querySelector('#menu-filter');
    if (filterButtons) {
        filterButtons.addEventListener('click', function(event) {
            // Only handle <a> element clicks
            if (!event.target.matches('a')) return;

            event.preventDefault();

            var filterValue = event.target.getAttribute('data-filter');

            // Update active button state
            var currentActive = filterButtons.querySelector('.active');
            if (currentActive) {
                currentActive.classList.remove('active');
            }
            event.target.classList.add('active');

            // Apply filter using arrange method
            iso.arrange({ filter: filterValue });
        });
    }
},
```

Key changes from v2 jQuery to v3 vanilla JS:
- `$(window).on('load')` wrapper removed (not needed, script runs at end of body)
- `$('.work-filter')` becomes `document.querySelector('.work-filter')`
- `$container.isotope({...})` becomes `new Isotope(grid, {...})`
- `animationOptions.duration: 750` becomes `transitionDuration: '0.75s'`
- `$filter.find('a').on('click')` becomes `addEventListener('click')` with delegation
- `$(this).attr('data-filter')` becomes `event.target.getAttribute('data-filter')`
- `$container.isotope({filter: x})` becomes `iso.arrange({filter: x})`
- Added `itemSelector: '.col-lg-4'` for explicit item targeting
- Added `percentPosition: true` for responsive layouts
